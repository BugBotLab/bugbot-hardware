# BugBot_NFCfloor  JLC 2-layer FR4 1.6 mm, bare board (no assembly): the floor outline with the wheel windows and ball
#                  notch cut out, a 3-turn loop round the edge on the top copper (against the floor), 2 plated holes at
#                  the centre for the cable's 2 wires (hand-soldered), 3 holes for the existing nylon clamp screws (M2 x 5).
#                  Replaces BugBot_NFCflex (and BugBot_NFCcoil); the reader board (BugBot_NFC) is unchanged.
#                  + underglow (inner ring, chosen 2026-10-22; positions from kicad/lib/inner_ring_geom.py): 8 x
#                  WS2812B-2020-V6 on the BOTTOM face on a circle r 9.5, 40 deg apart (the empty slot is where the NFC wires
#                  cross), 100 nF beside each; 74AHCT1G125 level shifter, 33 R, 10 uF (bottom); JST-SH 3 J2 (top, up through the floor opening).
#                  5 V / GND lanes (top copper) follow the LED line as one open chain: no copper ring inside the NFC loop.
#                  All routed here (Freerouting hangs on the long lanes): a fixed pattern per LED, see the underglow block.

# ---- underglow, routed by hand (Freerouting hangs on this board). Every LED gets the same pattern in its own frame
# (u along the data direction, v to the left): top copper = 5 V lane, GND lane (+-0.9) and the data line (centre);
# bottom = the LED, its 100 nF across the line at u -2.25, and short links to 5 vias:
#   Vc (0, 0) data in (centre of the LED, between its pads), Vd (+2.0, 0) data out, Vg1 (+1.6, GND side) for the LED's
#   GND pad, Vp / Vg2 (-3.0, -+0.9) for the cap's 5 V / GND pads; the LED's VDD pad goes to the cap's 5 V pad.
def bottom(f):
    f.SetLayerAndFlip(pcbnew.B_Cu)
    return f
PATH = [tuple(p) for p in UG["path"]] if UG.get("mode") == "inner" else [tuple(p) for p in UG["feed"]] + [tuple(p) for p in UG["bus"][1:]]
R_RING = UG.get("R", 0)
def tangent_at(pt):
    """the path's direction at pt, from points 0.5-1.5 mm either side along it (the data direction)"""
    i = min(range(len(PATH)), key=lambda k: math.dist(PATH[k], pt))
    j0 = i
    while j0 > 0 and math.dist(PATH[j0], pt) < 0.8: j0 -= 1
    j1 = i
    while j1 < len(PATH) - 1 and math.dist(PATH[j1], pt) < 0.8: j1 += 1
    a, b = PATH[j0], PATH[j1]
    L_ = math.dist(a, b); return ((b[0] - a[0]) / L_, (b[1] - a[1]) / L_), i
def walk(i, ds):
    """point and unit tangent ds mm along PATH from vertex i (negative: backwards)"""
    step = 1 if ds >= 0 else -1; left = abs(ds); j = i
    while True:
        k = j + step
        if k < 0 or k >= len(PATH):
            a, b = (PATH[j - 1], PATH[j]) if step > 0 else (PATH[j], PATH[j + 1]); L_ = math.dist(a, b)
            t = ((b[0] - a[0]) / L_, (b[1] - a[1]) / L_); return (PATH[j][0] + step * left * t[0], PATH[j][1] + step * left * t[1]), t
        seg = math.dist(PATH[j], PATH[k])
        if seg >= left:
            f_ = left / seg if seg else 0; q = (PATH[j][0] + (PATH[k][0] - PATH[j][0]) * f_, PATH[j][1] + (PATH[k][1] - PATH[j][1]) * f_)
            a, b = (PATH[j], PATH[k]) if step > 0 else (PATH[k], PATH[j]); L_ = math.dist(a, b)
            return q, ((b[0] - a[0]) / L_, (b[1] - a[1]) / L_)
        left -= seg; j = k
def frame(pt):
    (tx, ty), i = tangent_at(pt)
    def Q(u, v):
        q, t = walk(i, u)
        return (q[0] - v * t[1], q[1] + v * t[0])
    return Q, (tx, ty), i
def local(fr_pt, t, q):                                        # robot xy -> (u, v)
    dx, dy = q[0] - fr_pt[0], q[1] - fr_pt[1]
    return (dx * t[0] + dy * t[1], -dx * t[1] + dy * t[0])
LEDS = []
SV = None                                                       # +1: GND lane on the left (+v), 5 V on the right
for i, l in enumerate(UG["leds"]):
    pt = (l["x"], l["y"]); P, t, idx = frame(pt)
    f = bottom(X.add_fp(f"D{i + 1}", pt[0], pt[1], 0))
    f.Models().clear(); m = pcbnew.FP_3DMODEL(); m.m_Filename = "${KIPRJMOD}/../lib/3dmodels/LED_WS2812B-2020_PLCC4_2.0x2.0mm.step"; f.Models().push_back(m)
    ok = None
    for a in range(0, 360, 90):
        f.SetOrientationDegrees(math.degrees(math.atan2(t[1], t[0])) + a)
        pd = {p.GetNumber(): local(pt, t, X.xy(p)) for p in f.Pads()}
        if pd["3"][0] < 0 and pd["4"][0] < 0 and pd["1"][0] > 0 and pd["2"][0] > 0:      # DIN + VDD upstream, DOUT + GND downstream
            ok = (a, pd); break
    assert ok, f"D{i + 1}: no orientation with DIN upstream"
    sv = 1 if ok[1]["2"][1] > 0 else -1                         # the GND pad's side
    SV = sv if SV is None else SV
    assert sv == SV, "LED pad sides differ between LEDs"
    LEDS.append((i, pt, P, t, idx, f))
LANE5, LANEG = (UG["lane_a"], UG["lane_b"]) if SV > 0 else (UG["lane_b"], UG["lane_a"])   # lane_a is right of the path (-v)
X.track([tuple(p) for p in LANE5], "+5V", UG["lane_w"], pcbnew.F_Cu)
X.track([tuple(p) for p in LANEG], "GND", UG["lane_w"], pcbnew.F_Cu)
W_S, W_P, VIA_ = 0.2, 0.3, (0.55, 0.3)
padxy = lambda f, n: X.xy([p for p in f.Pads() if p.GetNumber() == n][0])
for i, pt, P, t, idx, f in LEDS:
    din, dout = f"D{i + 1}_DIN", (f"D{i + 2}_DIN" if i < len(LEDS) - 1 else None)
    cpt, ct = walk(idx, -2.25)
    cap = bottom(X.add_fp(f"C{i + 1}", *cpt, 0))
    for a in (90, -90):                                          # pads across the line, its 5 V pad on the 5 V side
        cap.SetOrientationDegrees(math.degrees(math.atan2(ct[1], ct[0])) + a)
        if local(cpt, ct, padxy(cap, "1"))[1] * SV < 0: break
    Vc, Vd, Vg1, Vp, Vg2 = P(0, 0), P(2.0, 0), P(1.6, 0.9 * SV), P(-3.0, -0.9 * SV), P(-3.0, 0.9 * SV)
    X.via(*Vc, din, *VIA_); X.via(*Vg1, "GND", *VIA_); X.via(*Vp, "+5V", *VIA_); X.via(*Vg2, "GND", *VIA_)
    X.track([padxy(f, "3"), Vc], din, W_S, pcbnew.B_Cu)
    X.track([padxy(f, "2"), Vg1], "GND", W_P, pcbnew.B_Cu)
    X.track([padxy(f, "4"), padxy(cap, "1")], "+5V", W_P, pcbnew.B_Cu)
    X.track([padxy(cap, "1"), Vp], "+5V", W_P, pcbnew.B_Cu)
    X.track([padxy(cap, "2"), Vg2], "GND", W_P, pcbnew.B_Cu)
    if dout:
        X.via(*Vd, dout, *VIA_)
        X.track([padxy(f, "1"), Vd], dout, W_S, pcbnew.B_Cu)
        nxt = LEDS[i + 1]
        seg = [p for p in PATH[idx:nxt[4] + 1] if math.dist(p, pt) > 2.05 and math.dist(p, nxt[1]) > 0.3]
        X.track([Vd] + seg + [nxt[1]], dout, W_S, pcbnew.F_Cu)          # data, top centre, to the next LED's Vc
X.track([PATH[0]] + [p for p in PATH[1:LEDS[0][4] + 1] if math.dist(p, LEDS[0][1]) > 0.3] + [LEDS[0][1]], "D1_DIN", W_S, pcbnew.F_Cu)   # start via -> D1

# ---- centre (inner-ring version): J2 (top) -> 3 vias -> bottom routes out to the chain start at the ring's gap (UG
# start_deg), where they come up onto the top lanes / data line. The NFC wires cross the gap on the TOP copper only, so
# the bottom routes pass under them. Order at the gap, north to west: outer lane (130 deg), data (137), inner lane (144).
# The level shifter is on the main board (it sends 5 V-level data); C13 10 uF sits under J2.
ring_pt = lambda r, deg: (r * math.cos(math.radians(deg)), r * math.sin(math.radians(deg)))
l5, lg = tuple(LANE5[0]), tuple(LANEG[0])
outer_net = "+5V" if math.hypot(*l5) > math.hypot(*lg) else "GND"
inner_net = "GND" if outer_net == "+5V" else "+5V"
NETPAD = {"+5V": "1", "D1_DIN": "2", "GND": "3"}
j2 = X.add_fp("J2", 0.0, -1.2, 0)
for a in (0, 180):                                               # the outer lane's pin on the left (-x): it becomes the eastmost route
    j2.SetOrientationDegrees(a)
    jp = {n: padxy(j2, n) for n in ("1", "2", "3")}
    if jp[NETPAD[outer_net]][0] < jp["2"][0]: break
for p in j2.Pads():
    if p.GetNumber() == "MP": p.SetNetCode(0)
yv = min(p[1] for p in jp.values()) - 1.23
VIAS = {n: (jp[NETPAD[n]][0], yv) for n in NETPAD}
for n_, v_ in VIAS.items():
    X.via(*v_, n_, *VIA_); X.track([jp[NETPAD[n_]], v_], n_, W_S if n_ == "D1_DIN" else W_P, pcbnew.F_Cu)
# C13 under J2, north of the vias: between the 5 V and GND vias
c13 = bottom(X.add_fp("C13", 0.0, yv + 1.35, 0))
cp = {n: padxy(c13, n) for n in ("1", "2")}
if (cp["1"][0] - cp["2"][0]) * (VIAS["+5V"][0] - VIAS["GND"][0]) < 0:
    c13.SetOrientationDegrees(180); cp = {n: padxy(c13, n) for n in ("1", "2")}
X.track([VIAS["+5V"], (VIAS["+5V"][0], cp["1"][1]), cp["1"]], "+5V", W_P, pcbnew.B_Cu)
X.track([VIAS["GND"], (VIAS["GND"][0], cp["2"][1]), cp["2"]], "GND", W_P, pcbnew.B_Cu)
# the 3 start vias on the start radial
SV_ = {outer_net: ring_pt(R_RING + 0.9, UG["start_deg"]), "D1_DIN": ring_pt(R_RING, UG["start_deg"]), inner_net: ring_pt(R_RING - 0.9, UG["start_deg"])}
for n_, v_ in SV_.items(): X.via(*v_, n_, *VIA_)
def arc(r, d0, d1, n=14): return [ring_pt(r, d0 + (d1 - d0) * k / n) for k in range(n + 1)]
# bottom routes: each via drops south, runs west, then north (the leftmost via -> the eastmost vertical), then out to the gap
order = sorted(VIAS, key=lambda n: VIAS[n][0])                     # left to right
for k, n_ in enumerate(order):
    v_ = VIAS[n_]
    yk, xk = yv - 0.75 - 0.8 * k, -3.0 - 0.9 * k                   # stagger: no crossings
    deg = {outer_net: UG["start_deg"] - 14, "D1_DIN": UG["start_deg"] - 7, inner_net: UG["start_deg"]}[n_]
    rk = {outer_net: R_RING + 0.9, "D1_DIN": R_RING, inner_net: R_RING - 0.9}[n_]
    r_in = 6.4 + 0.7 * (2 - k)
    ytop = 0.4 + 1.2 * k
    q = ring_pt(r_in, deg)                                          # north to the radial's height, then west onto it: parallel, no crossings
    pts = [v_, (v_[0], yk), (xk, yk), (xk, q[1]), q, ring_pt(rk, deg)] + arc(rk, deg, UG["start_deg"])[1:]
    X.track(pts, n_, W_S if n_ == "D1_DIN" else W_P, pcbnew.B_Cu)

X.track([tuple(p) for p in GEO["spiral"]], "COIL", PR["TW"], pcbnew.F_Cu)
X.via(*GEO["vias"][0], "COIL", *PR["via"]); X.via(*GEO["vias"][1], "COIL", *PR["via"])
X.track([tuple(p) for p in GEO["bottom"]], "COIL", PR["LW"], pcbnew.B_Cu)
la, lb = [tuple(p) for p in GEO["lead_a"]], [tuple(p) for p in GEO["lead_b"]]
pa = min(pads, key=lambda n: math.dist(pads[n], la[-1])); pb = "2" if pa == "1" else "1"
X.track(la[:-1] + [pads[pa]], "COIL", PR["LW"], pcbnew.F_Cu)
X.track(lb[:-1] + [pads[pb]], "COIL", PR["LW"], pcbnew.F_Cu)
for txt, (tx, ty), layer, sz in (("NFC wires", (0, 5.0), pcbnew.B_SilkS, 0.8), ("BugBot NFC floor", (-15, -24), pcbnew.B_SilkS, 1.2)):
    t = pcbnew.PCB_TEXT(X.board); t.SetText(txt); t.SetPosition(X.P(tx, ty)); t.SetLayer(layer); t.SetMirrored(True)
    t.SetTextSize(pcbnew.VECTOR2I(mm(sz), mm(sz))); t.SetTextThickness(mm(0.15)); X.board.Add(t)

          open(f"{X.out}/mcad.json", "w"), indent=1)
